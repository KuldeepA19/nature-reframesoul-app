// State Management
let state = {
    mood: 'neutral',
    plants: JSON.parse(localStorage.getItem('gardenData')) || []
};

// 1. Navigation
function changeScreen(screenId) {
    document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
    document.getElementById(screenId).classList.add('active');
}

// 2. Mood Handling
function setMood(m) {
    state.mood = m;
    const colors = {
        angry: '#fff5f5',
        stressed: '#f0f7ff',
        lonely: '#f6fff6'
    };
    document.body.style.backgroundColor = colors[m] || '#f0f4f0';
    document.getElementById('sync-text').innerText = `Syncing with your ${m} heart...`;
}

// 3. Planting & Reframing
function handleReframe() {
    const input = document.getElementById('vent-input');
    if (!input.value.trim()) return alert("Please share a thought to plant.");

    // Pick a random plant
    const emojis = ['🌱', '🌿', '🌸', '🌼', '🍀', '🌳'];
    const plant = emojis[Math.floor(Math.random() * emojis.length)];
    
    state.plants.push(plant);
    localStorage.setItem('gardenData', JSON.stringify(state.plants));

    renderGarden();
    input.value = '';
    
    // Quick vibration for feel
    if (navigator.vibrate) navigator.vibrate(50);
}

function renderGarden() {
    const view = document.getElementById('garden-view');
    view.innerHTML = state.plants.slice(-8).map(p => `<span>${p}</span>`).join('');
}

// Initialize on Load
window.onload = () => {
    renderGarden();
    document.getElementById('sync-text').innerText = "Vibe synchronized.";
};
